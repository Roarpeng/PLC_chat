              CORE ALGORITHM #1
              
老项目梯形图 (SimaticML FlgNet) 转 SCL 图论归约算法

            

            
          

          

            西门子 TIA Portal 导出的梯形图 XML 由 <Parts>（触点 `Contact`、线圈 `Coil`、功能块 `Call`）与 <Wires>（包含 `Powerrail`、`IdentCon`、`NameCon`、`Openbranch`）组成。不能直接把几千行 XML 丢给 LLM，必须先通过以下 **有向无环图 (DAG) 递归上溯算法** 精确还原能流布尔逻辑：
          

          
# =============================================================================
# 文件: apps/agent-core/compiler/lad_xml_to_scl.py
# 核心算法: 西门子 SimaticML (FlgNet) 梯形图能流归约转 SCL 编译器
# =============================================================================
import xml.etree.ElementTree as ET
from typing import Dict, List, Set

class FlgNetNode:
    def __init__(self, uid: str, part_name: str, negated: bool = False, operand: str = ""):
        self.uid = uid
        self.part_name = part_name  # 'Contact' | 'Coil' | 'Sr' | 'TON' | 'Gt' | 'Move'
        self.negated = negated      # True 表示常闭触点 (NC: --|/|-- )
        self.operand = operand      # 绑定的 PLC 变量符号名，如 '#i_bStart'
        self.in_pins: Dict[str, List[tuple]] = {}  # pin_name -> [(source_uid, source_pin)]

class SimaticLadToSclTranspiler:
    """
    将 TIA Openness 导出的梯形图 XML 确定性地编译为 SCL 抽象语法树 (AST)
    """
    def __init__(self, xml_content: str):
        self.root = ET.fromstring(xml_content)
        self.nodes: Dict[str, FlgNetNode] = {}
        self.powerrail_Connected_pins: Set[tuple] = set() # (uid, pin) 直接连在左母线上

    def parse_flgnet(self):
        # 1. 解析所有  元件 (触点、线圈、定时器、比较器)
        for part in self.root.findall(".//Parts/Part"):
            uid = part.attrib["UId"]
            name = part.attrib["Name"]
            negated = part.find("Negated") is not None
            operand = self._resolve_access_symbol(uid)
            self.nodes[uid] = FlgNetNode(uid, name, negated, operand)

        # 2. 解析所有  能流连线，构建逆向依赖图 (从输出端往左母线回溯)
        for wire in self.root.findall(".//Wires/Wire"):
            children = list(wire)
            source = children[0]
            targets = children[1:]

            if source.tag == "Powerrail":
                # 该能流线直接源自左侧母线 (TRUE)
                for tgt in targets:
                    if tgt.tag == "NameCon":
                        self.powerrail_Connected_pins.add((tgt.attrib["UId"], tgt.attrib["Name"]))
            elif source.tag == "NameCon":
                src_uid, src_pin = source.attrib["UId"], source.attrib["Name"]
                for tgt in targets:
                    if tgt.tag == "NameCon":
                        t_uid, t_pin = tgt.attrib["UId"], tgt.attrib["Name"]
                        self.nodes[t_uid].in_pins.setdefault(t_pin, []).append((src_uid, src_pin))

    def resolve_rung_expression(self, uid: str, in_pin: str = "in") -> str:
        """
        【核心递归算法】从任意元件的输入管脚回溯到左母线，自动归约串并联逻辑：
        - 同一管脚收到多条 Wire 汇入 (Openbranch) -> 并联 OR 关系
        - 触点沿着 in -> out 级联传递 -> 串联 AND 关系
        """
        if (uid, in_pin) in self.powerrail_Connected_pins:
            return "TRUE"

        upstream_list = self.nodes[uid].in_pins.get(in_pin, [])
        if not upstream_list:
            return "FALSE"

        branch_exprs = []
        for src_uid, src_pin in upstream_list:
            src_node = self.nodes[src_uid]
            # 递归求取上游元件自身的输入能流表达式
            prev_power = self.resolve_rung_expression(src_uid, "in")

            if src_node.part_name == "Contact":
                contact_sym = f"NOT {src_node.operand}" if src_node.negated else src_node.operand
                expr = contact_sym if prev_power == "TRUE" else f"({prev_power} AND {contact_sym})"
                branch_exprs.append(expr)
            elif src_node.part_name in ("Gt", "Lt", "Eq"):
                op_map = {"Gt": ">", "Lt": "<", "Eq": "="}
                in1 = self._resolve_pin_operand(src_uid, "in1")
                in2 = self._resolve_pin_operand(src_uid, "in2")
                cmp_expr = f"({in1} {op_map[src_node.part_name]} {in2})"
                expr = cmp_expr if prev_power == "TRUE" else f"({prev_power} AND {cmp_expr})"
                branch_exprs.append(expr)

        # 多条支路汇合即为并联 (OR)
        if len(branch_exprs) == 1:
            return branch_exprs[0]
        return "(" + " OR ".join(branch_exprs) + ")"

    def emit_scl_statements(self) -> List[str]:
        """遍历所有末端执行元件 (Coil / TON / Move)，生成标准 SCL 语句"""
        scl_lines = []
        for uid, node in self.nodes.items():
            if node.part_name == "Coil":
                rung_cond = self.resolve_rung_expression(uid, "in")
                scl_lines.append(f"{node.operand} := {rung_cond};")
            elif node.part_name == "TON":
                rung_cond = self.resolve_rung_expression(uid, "IN")
                pt_val = self._resolve_pin_operand(uid, "PT")
                scl_lines.append(f"{node.operand}(IN := {rung_cond}, PT := {pt_val});")
            elif node.part_name == "Move":
                en_cond = self.resolve_rung_expression(uid, "en")
                in_val = self._resolve_pin_operand(uid, "in")
                out_var = self._resolve_pin_operand(uid, "out1")
                scl_lines.append(f"IF {en_cond} THEN\n    {out_var} := {in_val};\nEND_IF;")
        return scl_lines

        


        
        

          

            

              CORE ALGORITHM #2
              
LangGraph 多智能体编排框架与编译自愈状态机

            

            
          

          

            无论是“新工艺正向生成”还是“老项目 LAD 逆向重构”，都在统一的 **LangGraph 状态图** 中运行。一旦 C# Openness Bridge 返回编译错误 (`error_count > 0`)，状态机自动路由至 `compiler_self_healer` 节点进行定位修复：
          

          
# =============================================================================
# 文件: apps/agent-core/workflow/chat2plc_graph.py
# 框架: LangGraph 工业级状态机 (支持正向工艺生成 + 老项目 LAD 转 SCL + 编译自愈)
# =============================================================================
from typing import TypedDict, List, Literal, Optional
from langgraph.graph import StateGraph, END
import httpx

class Chat2PlcState(TypedDict):
    mode: Literal["forward_process", "legacy_lad_to_scl"]
    user_prompt: str                   # 用户的自然语言工艺描述
    block_name: str                    # 目标 PLC 块名称，如 "FB_MixingTank_Ctrl"
    legacy_lad_xml: Optional[str]      # 从老项目导出的 SimaticML XML
    process_ir: Optional[dict]         # 工艺中间表示层 (I/O + 联锁 + 步序)
    scl_code: str                      # 当前生成的符合西门子规范的 SCL 源码
    compile_diagnostics: List[dict]    # 博途编译器返回的精确行号报错列表
    heal_retry_count: int              # 自愈重试计数器 (上限 3 次)
    status: str

# ----------------- 核心智能体节点定义 -----------------

async def legacy_lad_decompiler_node(state: Chat2PlcState) -> Chat2PlcState:
    """节点 A1: 针对老项目，调用 C# Bridge 导出 LAD XML 并执行 DAG 归约 + 语义重构"""
    async with httpx.AsyncClient() as client:
        resp = await client.get(f"http://localhost:5005/api/openness/export-xml/{state['block_name']}")
        xml_str = resp.json()["simaticMlXml"]

    # 1. 使用确定性图算法提取布尔代数骨架 (100% 保证逻辑等价)
    transpiler = SimaticLadToSclTranspiler(xml_str)
    transpiler.parse_flgnet()
    raw_scl_statements = transpiler.emit_scl_statements()

    # 2. 调用 LLM 将分散的布尔自锁/步序标志位重写为清晰的结构化 SCL 并补齐中文工艺注释
    refactored_scl = await llm_semantic_refactor(raw_scl_statements, state["user_prompt"])
    return {**state, "legacy_lad_xml": xml_str, "scl_code": refactored_scl, "heal_retry_count": 0}

async def process_and_safety_node(state: Chat2PlcState) -> Chat2PlcState:
    """节点 A2: 针对新工艺，生成 Process-IR 并强制注入急停、传感器合理性与阀门超时联锁"""
    ir = await generate_process_ir_with_safety_guardrails(state["user_prompt"])
    scl = await generate_siemens_style_scl_from_ir(ir)
    return {**state, "process_ir": ir, "scl_code": scl, "heal_retry_count": 0}

async def openness_compile_verify_node(state: Chat2PlcState) -> Chat2PlcState:
    """节点 B: 调用本地 C# STA 守护进程导入 .scl 源文件并触发 TIA Portal 真实编译"""
    async with httpx.AsyncClient(timeout=30.0) as client:
        resp = await client.post("http://localhost:5005/api/openness/compile-scl", json={
            "blockName": state["block_name"],
            "sclContent": state["scl_code"]
        })
        diag = resp.json() # { "success": bool, "errorCount": int, "messages": [...] }

    return {
        **state,
        "compile_diagnostics": diag["messages"],
        "status": "compiled_ok" if diag["errorCount"] == 0 else "compile_failed"
    }

async def compiler_self_healer_node(state: Chat2PlcState) -> Chat2PlcState:
    """节点 C: 编译自愈智能体 - 根据博途返回的精确行号与报错描述修复 SCL 语法/类型不匹配"""
    fixed_scl = await llm_fix_scl_with_diagnostics(
        original_scl=state["scl_code"],
        compiler_errors=state["compile_diagnostics"]
    )
    return {**state, "scl_code": fixed_scl, "heal_retry_count": state["heal_retry_count"] + 1}

# ----------------- 路由条件与状态图装配 -----------------

def route_after_compile(state: Chat2PlcState) -> Literal["healer", "finish"]:
    if state["status"] == "compiled_ok" or state["heal_retry_count"] >= 3:
        return "finish"
    return "healer"

workflow = StateGraph(Chat2PlcState)
workflow.add_node("legacy_decompiler", legacy_lad_decompiler_node)
workflow.add_node("forward_generator", process_and_safety_node)
workflow.add_node("tia_compiler", openness_compile_verify_node)
workflow.add_node("self_healer", compiler_self_healer_node)

workflow.set_conditional_entry_point(
    lambda s: "legacy_decompiler" if s["mode"] == "legacy_lad_to_scl" else "forward_generator"
)
workflow.add_edge("legacy_decompiler", "tia_compiler")
workflow.add_edge("forward_generator", "tia_compiler")
workflow.add_conditional_edges("tia_compiler", route_after_compile, {
    "healer": "self_healer",
    "finish": END
})
workflow.add_edge("self_healer", "tia_compiler")
chat2plc_app = workflow.compile()

              MODULE 1 · STA THREAD SAFETY
              
1. 单线程单元 (STA) 调度器与博途进程挂载

            

            
          

          
// =============================================================================
// 文件: services/TiaOpennessBridge.Host/Core/StaOpennessDispatcher.cs
// 作用: 强制所有针对 Siemens.Engineering.dll 的 API 调用在同一个 STA 线程顺序执行
// =============================================================================
using System;
using System.Collections.Concurrent;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using Siemens.Engineering;
using Siemens.Engineering.HW;
using Siemens.Engineering.SW;

public sealed class StaOpennessDispatcher : IDisposable
{
    private readonly BlockingCollection _workQueue = new BlockingCollection();
    private readonly Thread _staThread;
    public TiaPortal CurrentPortal { get; private set; }
    public Project ActiveProject { get; private set; }
    public PlcSoftware ActivePlc { get; private set; }

    public StaOpennessDispatcher()
    {
        _staThread = new Thread(StaPumpLoop)
        {
            Name = "TIA_Openness_STA_Worker",
            IsBackground = true
        };
        // 【关键】必须设置为 STA (Single-Threaded Apartment)，否则 Openness COM 接口直接抛异常
        _staThread.SetApartmentState(ApartmentState.STA);
        _staThread.Start();
    }

    private void StaPumpLoop()
    {
        foreach (var action in _workQueue.GetConsumingEnumerable())
        {
            action();
        }
    }

    // 供 WebAPI Controller 异步调用的线程安全包装器
    public Task InvokeAsync(Func opennessFunc)
    {
        var tcs = new TaskCompletionSource();
        _workQueue.Add(() =>
        {
            try
            {
                T result = opennessFunc();
                tcs.SetResult(result);
            }
            catch (EngineeringException ex)
            {
                tcs.SetException(new Exception($"[TIA Openness Error] {ex.Message}", ex));
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        return tcs.Task;
    }

    // 附加到当前正在运行的 TIA Portal 窗口并定位第一个 PLC_1
    public Task AttachToRunningTiaAsync() => InvokeAsync(() =>
    {
        var processes = TiaPortal.GetProcesses();
        var targetProc = processes.FirstOrDefault()
            ?? throw new InvalidOperationException("未检测到正在运行的 TIA Portal 进程！");

        CurrentPortal = targetProc.Attach();
        ActiveProject = CurrentPortal.Projects.FirstOrDefault()
            ?? throw new InvalidOperationException("TIA Portal 中尚未打开任何工程项目！");

        // 遍历硬件设备寻找 PlcSoftware 容器
        foreach (Device device in ActiveProject.Devices)
        {
            foreach (DeviceItem item in device.DeviceItems)
            {
                var swContainer = item.GetService();
                if (swContainer?.Software is PlcSoftware plc)
                {
                    ActivePlc = plc;
                    return $"成功挂载项目 [{ActiveProject.Name}] -> 控制器 [{plc.Name}]";
                }
            }
        }
        throw new InvalidOperationException("工程中未找到 S7-1200/1500 PLC 软件容器！");
    });

    public void Dispose() => _workQueue.CompleteAdding();
}

// =============================================================================
// 文件: services/TiaOpennessBridge.Host/Services/PlcBlockBridgeService.cs
// 作用: 1) 将存量老项目的 LAD 块导出为 SimaticML XML 供 Python 逆向转 SCL
//       2) 将 AI 生成的 SCL 注入博途并提取精确到行号的编译错误树
// =============================================================================
using System.Collections.Generic;
using System.IO;
using System.Text;
using System.Threading.Tasks;
using Siemens.Engineering;
using Siemens.Engineering.Compiler;
using Siemens.Engineering.SW.Blocks;
using Siemens.Engineering.SW.ExternalSources;

public class PlcBlockBridgeService
{
    private readonly StaOpennessDispatcher _sta;
    public PlcBlockBridgeService(StaOpennessDispatcher sta) => _sta = sta;

    // 【核心接口 1】导出老项目中的梯形图 (LAD) 块为 SimaticML XML
    public Task ExportLegacyLadBlockToXmlAsync(string blockName) => _sta.InvokeAsync(() =>
    {
        PlcBlock block = _sta.ActivePlc.BlockGroup.Blocks.Find(blockName)
            ?? throw new FileNotFoundException($"在博途中未找到块: {blockName}");

        // 确认编程语言 (ProgrammingLanguage.LAD / FBD / SCL)
        string tempXmlPath = Path.Combine(Path.GetTempPath(), $"{blockName}_LadExport.xml");
        if (File.Exists(tempXmlPath)) File.Delete(tempXmlPath);

        // 调用 Openness Export 导出包含  拓扑结构的完整 XML
        block.Export(new FileInfo(tempXmlPath), ExportOptions.WithDefaults);
        return File.ReadAllText(tempXmlPath, Encoding.UTF8);
    });

    // 【核心接口 2】注入 SCL 源码 -> 生成块 -> 编译 -> 递归提取报错行号
    public Task InjectSclAndCompileAsync(string blockName, string sclCode) => _sta.InvokeAsync(() =>
    {
        var plc = _sta.ActivePlc;
        string tempSclFile = Path.Combine(Path.GetTempPath(), $"{blockName}.scl");
        File.WriteAllText(tempSclFile, sclCode, new UTF8Encoding(encoderShouldEmitUTF8Identifier: true));

        // 1. 清理同名旧外部源文件并重新注册
        PlcExternalSourceSystemGroup extGroup = plc.ExternalSourceGroup;
        PlcExternalSource existingSrc = extGroup.ExternalSources.Find($"{blockName}.scl");
        existingSrc?.Delete();

        PlcExternalSource newSrc = extGroup.ExternalSources.CreateFromFile($"{blockName}.scl", tempSclFile);

        // 2. 从 .scl 文本生成博途逻辑块 (KeepOnError 保证即便有小错也能进入编译器获取行号)
        newSrc.GenerateBlockFromSource(GenerateBlockOption.KeepOnError);

        // 3. 针对刚生成的块触发增量编译 (ICompilable)
        PlcBlock generatedBlock = plc.BlockGroup.Blocks.Find(blockName);
        ICompilable compiler = (generatedBlock as IEngineeringServiceProvider)?.GetService()
                               ?? plc.GetService();
        CompilerResult result = compiler.Compile();

        // 4. 递归解析树状编译消息，回传给 Python Self-Healer Agent
        var dto = new CompileDiagnosticDto
        {
            Success = result.State != CompilerResultState.Error,
            ErrorCount = result.ErrorCount,
            WarningCount = result.WarningCount,
            Messages = new List()
        };
        ExtractMessagesRecursive(result.Messages, dto.Messages);
        return dto;
    });

    // 递归遍历博途的多层 CompilerResultMessage 树
    private void ExtractMessagesRecursive(CompilerResultMessageComposition messages, List output)
    {
        foreach (CompilerResultMessage msg in messages)
        {
            if (!string.IsNullOrWhiteSpace(msg.Description) &&
                (msg.State == CompilerResultState.Error || msg.State == CompilerResultState.Warning))
            {
                output.Add(new CompileMsgItem
                {
                    Path = msg.Path,               // 例如: "FB_MixingTank_Ctrl / Line 48"
                    State = msg.State.ToString(),  // "Error" | "Warning"
                    Description = msg.Description  // 例如: "参数 'PT' 的数据类型不匹配"
                });
            }
            if (msg.Messages != null && msg.Messages.Count > 0)
            {
                ExtractMessagesRecursive(msg.Messages, output);
            }
        }
    }
}
